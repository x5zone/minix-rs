#!/usr/bin/env python3
"""Phase 3 相对链接重定基（一次性工具，放 tmp/pre-migrate-snapshot-20261007/）。

要解决的问题：三棵树从 4 层搬到 2 层，stage 内部的兄弟链接（../02-stage-vm/x.md）深度不变照旧可用，
但跨出 stage 的链接（../../../concepts/README.md、../../README.md 这类）深度变了，
文本没变而解析目标变了——最坏的一种是「链接仍然可解析，但指到了另一个文件上」，断链检查抓不到。

算法（glm 附录 C 的规格，输入是映射表而不是旧树）：
  1. 用 path-map.tsv 建两张表：old→new 与 new→old。
  2. 对每个在范围内的新树 md 文件：由 new 路径反查它在迁移前的旧路径，得到「旧所在目录」。
  3. 对文件里每个相对链接：按旧所在目录解析成旧绝对目标。
     - 若该旧目标在映射表的旧路径集合里（= 它确实是被本次迁移搬走的一个真实文件）：
       正确目标 = 映射表给它的新路径，再算出「从新所在目录出发的相对路径」；与现文本不同则改写。
     - 若旧目标不在集合里（迁移前就指向不存在的东西）：不动，进报告，按用户裁决不顺手修历史断链。
  4. 跳过：fenced code block 内的行、外链、纯锚点、指向 evidence/ 与 .design/ 与 legacy-fork-bak/ 之外
     冻结区以外的一切正常处理（冻结区只是「文件本身不被改写」，指向它们的链接照样要重定基）。
  5. 只替换括号内的目标串，保留 #fragment 尾巴；逐行替换，行数不变。
"""
import posixpath
import re
import sys
from pathlib import Path, PurePosixPath

ROOT = Path("/home/xzhao/github/minix-rs")
MAP = ROOT / "migrate_notes_plan/pre-migrate-20261007/path-map.tsv"
APPLY = len(sys.argv) > 1 and sys.argv[1] == "--apply"

INLINE_LINK = re.compile(rb"(\]\()([^)\s#]+)(#[^)\s]*)?(\))")
FENCE = re.compile(rb"^\s*(```|~~~)")
EXTERNAL = (b"http://", b"https://", b"mailto:", b"ftp://", b"file://")


def load_map():
    old_to_new, new_to_old = {}, {}
    for line in MAP.read_text(encoding="utf-8").splitlines()[1:]:
        old, new, kind, rule = line.split("\t")
        old_to_new[old] = new
        new_to_old.setdefault(new, old)
    return old_to_new, new_to_old
