# VM 进程生命周期支持

## 模块整体定位

### 在系统中的作用

进程生命周期支持模块是 VM 服务器与 PM（进程管理器）协作的核心，负责：

1. **Fork 支持**: 创建子进程的地址空间副本
2. **Exit 支持**: 清理进程的内存资源
3. **Break 支持**: 调整进程的堆内存大小

## 核心数据结构

### 1. Fork 消息

**定义位置**: `servers/vm/fork.c`

```c
message *msg;  // 来自 PM 的 fork 请求

msg->VMF_ENDPOINT;     // 父进程端点
msg->VMF_SLOTNO;       // 子进程槽位号
msg->VMF_CHILD_ENDPOINT;  // 返回子进程端点
```

**字段说明**:

| 字段 | 类型 | 说明 |
|------|------|------|
| `VMF_ENDPOINT` | `endpoint_t` | 父进程端点 |
| `VMF_SLOTNO` | `int` | 子进程槽位号 |
| `VMF_CHILD_ENDPOINT` | `endpoint_t` | 返回的子进程端点 |

### 2. Exit 消息

**定义位置**: `servers/vm/exit.c`

```c
message *msg;  // 来自 PM 的 exit 请求

msg->VME_ENDPOINT;     // 退出进程的端点
```

**字段说明**:

| 字段 | 类型 | 说明 |
|------|------|------|
| `VME_ENDPOINT` | `endpoint_t` | 退出进程的端点 |

### 3. Break 消息

**定义位置**: `servers/vm/break.c`

```c
message *msg;  // 来自进程的 brk 请求

msg->m_source;           // 调用者端点
msg->m_lc_vm_brk.addr;   // 新的堆顶地址
```

**字段说明**:

| 字段 | 类型 | 说明 |
|------|------|------|
| `m_source` | `endpoint_t` | 调用者端点 |
| `m_lc_vm_brk.addr` | `vir_bytes` | 新的堆顶地址 |

**关键代码**:

```c
int do_fork(message *msg)
{
    int r, proc, childproc;
    struct vmproc *vmp, *vmc;
    pt_t origpt;
    vir_bytes msgaddr;

    SANITYCHECK(SCL_FUNCTIONS);

    // 步骤 1: 验证参数
    if (vm_isokendpt(msg->VMF_ENDPOINT, &proc) != OK) {
        printf("VM: bogus endpoint VM_FORK %d\n", msg->VMF_ENDPOINT);
        SANITYCHECK(SCL_FUNCTIONS);
        return EINVAL;
    }

    childproc = msg->VMF_SLOTNO;
    if (childproc < 0 || childproc >= NR_PROCS) {
        printf("VM: bogus slotno VM_FORK %d\n", msg->VMF_SLOTNO);
        SANITYCHECK(SCL_FUNCTIONS);
        return EINVAL;
    }

    vmp = &vmproc[proc];      // 父进程
    vmc = &vmproc[childproc]; // 子进程
    assert(vmc->vm_slot == childproc);

    // 步骤 2: 复制进程结构
    origpt = vmc->vm_pt;
    *vmc = *vmp;
    vmc->vm_slot = childproc;
    region_init(&vmc->vm_regions_avl);
    vmc->vm_endpoint = NONE;
    vmc->vm_pt = origpt;

    // 步骤 3: 创建新页表
    if (pt_new(&vmc->vm_pt) != OK) {
        return ENOMEM;
    }

    SANITYCHECK(SCL_DETAIL);

    // 步骤 4: 复制地址空间
    if (map_proc_copy(vmc, vmp) != OK) {
        printf("VM: fork: map_proc_copy failed\n");
        pt_free(&vmc->vm_pt);
        return ENOMEM;
    }

    // 步骤 5: 设置标志位
    vmc->vm_flags &= VMF_INUSE;

    // 步骤 6: ACL 处理
    acl_fork(vmc);

    // 步骤 7: 通知内核
    if ((r = sys_fork(vmp->vm_endpoint, childproc,
        &vmc->vm_endpoint, PFF_VMINHIBIT, &msgaddr)) != OK) {
        panic("do_fork can't sys_fork: %d", r);
    }

    // 步骤 8: 绑定页表
    if ((r = pt_bind(&vmc->vm_pt, vmc)) != OK)
        panic("fork can't pt_bind: %d", r);

    // 步骤 9: 处理消息内存
    {
        vir_bytes vir;
        vir = msgaddr;
        if (handle_memory_once(vmc, vir, sizeof(message), 1) != OK)
            panic("do_fork: handle_memory for child failed\n");
        vir = msgaddr;
        if (handle_memory_once(vmp, vir, sizeof(message), 1) != OK)
            panic("do_fork: handle_memory for parent failed\n");
    }

    // 步骤 10: 返回结果
    msg->VMF_CHILD_ENDPOINT = vmc->vm_endpoint;

    SANITYCHECK(SCL_FUNCTIONS);
    return OK;
}
```

```c
int do_exit(message *msg)
{
    int proc;
    struct vmproc *vmp;

    SANITYCHECK(SCL_FUNCTIONS);

    // 步骤 1: 验证参数
    if (vm_isokendpt(msg->VME_ENDPOINT, &proc) != OK) {
        printf("VM: bogus endpoint VM_EXIT %d\n", msg->VME_ENDPOINT);
        return EINVAL;
    }
    vmp = &vmproc[proc];

    // 步骤 2: 检查状态
    if (!(vmp->vm_flags & VMF_EXITING)) {
        printf("VM: unannounced VM_EXIT %d\n", msg->VME_ENDPOINT);
        return EINVAL;
    }
    
    // 步骤 3: 处理 VM 实例
    if (vmp->vm_flags & VMF_VM_INSTANCE) {
        vmp->vm_flags &= ~VMF_VM_INSTANCE;
        num_vm_instances--;
    }

    // 步骤 4: 释放内存
    SANITYCHECK(SCL_DETAIL);
    free_proc(vmp);
    SANITYCHECK(SCL_DETAIL);

    // 步骤 5: 清理进程结构
    clear_proc(vmp);

    SANITYCHECK(SCL_FUNCTIONS);
    return OK;
}
```

```c
int do_brk(message *msg)
{
    int proc;

    if (vm_isokendpt(msg->m_source, &proc) != OK) {
        printf("VM: bogus endpoint VM_BRK %d\n", msg->m_source);
        return EINVAL;
    }

    return real_brk(&vmproc[proc], (vir_bytes) msg->m_lc_vm_brk.addr);
}

int real_brk(struct vmproc *vmp, vir_bytes v)
{
    if (map_region_extend_upto_v(vmp, v) == OK) {
        return OK;
    }

    return ENOMEM;
}
```

## 关键机制拆解

### 1. Fork 的 COW 机制

**原理**: 使用写时复制共享物理页

```
┌─────────────────────────────────────────────────────────────────────────────┐
│                          Fork 的 COW 机制                                    │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                             │
│   Fork 前:                                                                  │
│   ┌─────────────────────────────────────────────────────────────────────┐   │
```

### 2. Exit 的两阶段清理

**原理**: 分两阶段清理进程资源

```

│                          Exit 的两阶段清理                                   │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                             │
│   阶段 1: VM_WILLEXIT (预告)                                                │
│   ┌─────────────────────────────────────────────────────────────────────┐   │
│   │ - PM 通知 VM 进程即将退出                                            │   │
│   │ - VM 设置 VMF_EXITING 标志                                           │   │
│   │ - 进程仍在运行                                                       │   │
│   │                                                                     │   │
│   │ int do_willexit(message *msg)                                        │   │
│   │ {                                                                    │   │
│   │     vmp->vm_flags |= VMF_EXITING;                                    │   │
│   │     return OK;                                                       │   │
│   │ }                                                                    │   │
│   └─────────────────────────────────────────────────────────────────────┘   │
│                                    │                                        │
│                                    ↓                                        │
│   阶段 2: VM_EXIT (实际清理)                                                │
│   ┌─────────────────────────────────────────────────────────────────────┐   │
│   │ - PM 通知 VM 进程已退出                                              │   │
│   │ - VM 清理所有内存资源                                                │   │
│   │ - 进程已停止运行                                                     │   │
│   │                                                                     │   │
│   │ int do_exit(message *msg)                                            │   │
│   │ {                                                                    │   │
│   │     free_proc(vmp);   // 释放内存                                    │   │
│   │     clear_proc(vmp);  // 清理结构                                    │   │
│   │     return OK;                                                       │   │
│   │ }                                                                    │   │
│   └─────────────────────────────────────────────────────────────────────┘   │
│                                                                             │
│   两阶段设计原因:                                                           │
│   ┌─────────────────────────────────────────────────────────────────────┐   │
│   │ 1. 避免竞态条件：确保进程已停止运行后再清理                           │   │
│   │ 2. 资源管理：可以提前准备资源清理                                     │   │
│   │ 3. 错误处理：可以在退出前检查状态                                     │   │
```

## 边界条件与特殊分支

### 1. Fork 失败处理

**代码位置**: `fork.c`

```c
if (map_proc_copy(vmc, vmp) != OK) {
    printf("VM: fork: map_proc_copy failed\n");
    pt_free(&vmc->vm_pt);  // 释放已创建的页表
    return ENOMEM;
}
```

**设计原因**: Fork 失败时需要清理已分配的资源。

### 2. Exit 状态检查

**代码位置**: `exit.c`

```c
if (!(vmp->vm_flags & VMF_EXITING)) {
    printf("VM: unannounced VM_EXIT %d\n", msg->VME_ENDPOINT);
    return EINVAL;
}
```

**设计原因**: 进程必须先调用 VM_WILLEXIT，防止意外退出。

### 3. Break 地址验证

**代码位置**: `break.c`

```c
int real_brk(struct vmproc *vmp, vir_bytes v)
{
    if (map_region_extend_upto_v(vmp, v) == OK) {
        return OK;
    }
    return ENOMEM;
}
```

**注意**: `map_region_extend_upto_v()` 会验证地址是否合法。

### 4. VM 实例计数

**代码位置**: `exit.c`

```c
if (vmp->vm_flags & VMF_VM_INSTANCE) {
    vmp->vm_flags &= ~VMF_VM_INSTANCE;
```

**设计原因**: 跟踪 VM 实例数量，用于系统管理。

## Rust 重构与设计改进建议

### 1. 类型系统改进

**当前问题**: 使用整数表示进程槽位和端点

```c
int childproc;  // 进程槽位
endpoint_t ep;  // 进程端点
```

**Rust 改进**:

```rust
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct ProcessSlot(pub usize);

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Endpoint(pub i32);

impl ProcessSlot {
    pub fn is_valid(&self) -> bool {
        self.0 < NR_PROCS
    }
}

impl Endpoint {
    pub fn is_valid(&self) -> bool {
        self.0 != NONE
    }
}
```

**优势**:
- 编译时类型检查
- 防止混淆槽位和端点
- 封装验证方法

### 2. 进程状态建模

**当前问题**: 使用标志位表示进程状态

```c
vmp->vm_flags |= VMF_EXITING;
```

**Rust 改进**:

```rust
enum ProcessState {
    Running,
    Exiting,
    Exited,
}

struct VmProcess {
    state: ProcessState,
    slot: ProcessSlot,
    endpoint: Endpoint,
    regions: RegionTree,
    page_table: PageTable,
}

impl VmProcess {
    fn transition(&mut self, new_state: ProcessState) -> Result<(), StateError> {
        match (&self.state, &new_state) {
            (ProcessState::Running, ProcessState::Exiting) => {
                self.state = new_state;
                Ok(())
            }
            (ProcessState::Exiting, ProcessState::Exited) => {
                self.state = new_state;
                Ok(())
            }
            _ => Err(StateError::InvalidTransition),
        }
    }
}
```

**优势**:
- 显式的状态转换
- 编译器检查状态完整性
- 防止非法状态转换

### 3. Fork 的所有权模型

**当前问题**: Fork 需要手动管理资源

```c
if (map_proc_copy(vmc, vmp) != OK) {
    pt_free(&vmc->vm_pt);
    return ENOMEM;
}
```

**Rust 改进**:

```rust
impl VmProcess {
    fn fork(&self, child_slot: ProcessSlot) -> Result<VmProcess, ForkError> {
        let mut child = VmProcess {
            state: ProcessState::Running,
            slot: child_slot,
            endpoint: Endpoint::new(),
            regions: RegionTree::new(),
            page_table: PageTable::new()?,
        };
        
        child.regions = self.regions.fork_cow()?;
        
        Ok(child)
    }
}

impl RegionTree {
    fn fork_cow(&self) -> Result<RegionTree, ForkError> {
        let mut new_tree = RegionTree::new();
        
        for region in &self.regions {
            let new_region = region.fork_cow()?;
            new_tree.insert(new_region)?;
        }
        
        Ok(new_tree)
    }
}
```

**优势**:
- 自动资源管理
- 所有权清晰
- 错误处理安全

### 4. Break 的内存管理

**当前问题**: Break 使用简单的地址扩展

```c
int real_brk(struct vmproc *vmp, vir_bytes v)
{
    if (map_region_extend_upto_v(vmp, v) == OK) {
        return OK;
    }
    return ENOMEM;
}
```

**Rust 改进**:

```rust
impl VmProcess {
    fn brk(&mut self, new_brk: VirtAddr) -> Result<VirtAddr, BrkError> {
        let current_brk = self.heap.end();
        
        if new_brk > current_brk {
            // 扩展堆
            let size = new_brk - current_brk;
            self.heap.extend(size)?;
        } else if new_brk < current_brk {
            // 收缩堆
            let size = current_brk - new_brk;
            self.heap.shrink(size)?;
        }
        
        Ok(self.heap.end())
    }
}

impl HeapRegion {
    fn extend(&mut self, size: usize) -> Result<(), MemoryError> {
        let new_end = self.end + size;
        
        // 检查是否会与栈冲突
        if new_end >= self.stack_start {
            return Err(MemoryError::AddressSpaceConflict);
        }
        
        // 扩展区域
        self.end = new_end;
        
        Ok(())
    }
}
```

**优势**:
- 类型安全的地址操作
- 显式的错误处理
- 自动边界检查

## 要点总结

1. **Fork 流程**
   - 复制进程结构
   - 创建新页表
   - 使用 COW 复制地址空间
   - 通知内核创建子进程

2. **Exit 流程**
   - 两阶段清理（WILLEXIT + EXIT）
   - 释放所有内存资源
   - 清理进程结构

3. **Break 流程**
   - 调整堆区域大小
   - 按需分配物理页
   - 支持扩展和收缩

4. **COW 机制**
   - Fork 时共享物理页
   - 写操作时复制
   - 提高内存效率

## 灾难预演

**场景 1: Fork 炸弹**

如果进程不断 fork：
```c
while (1) {
    fork();
}
```
- 进程数量指数增长
- 内存耗尽
- 系统崩溃

**预防**: 设置进程数量上限，限制用户进程数

**场景 2: Exit 时资源泄漏**

如果 Exit 未正确清理资源：
```c
void free_proc(struct vmproc *vmp) {
    // 忘记释放某些资源
    // map_free_proc(vmp);  // 忘记调用
}
```
- 内存泄漏
- 页表泄漏
- 系统资源耗尽

**预防**: 使用健全性检查，验证资源释放

**场景 3: Break 地址冲突**

如果 Break 地址与栈冲突：
```c
// 堆和栈向中间增长，最终相遇
heap_end = 0x7FFF0000;
stack_start = 0x7FFF0000;
```
- 地址空间冲突
- 内存访问错误
- 进程崩溃

**预防**: 检查地址边界，防止冲突

**场景 4: Fork 时页表创建失败**

如果 Fork 时无法创建页表：
```c
if (pt_new(&vmc->vm_pt) != OK) {
    return ENOMEM;  // 无法创建子进程
}
```
- Fork 失败
- 进程创建失败
- 用户操作失败

**预防**: 设置内存预留，确保关键操作有足够内存
