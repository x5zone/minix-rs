# VM 内存映射与内存类型

## 模块整体定位

### 在系统中的作用

内存映射与内存类型模块是 VM 服务器的核心功能抽象层，负责：

1. **mmap 系统调用**: 实现内存映射接口
2. **内存类型抽象**: 统一的内存类型接口，支持多种内存类型
3. **匿名内存**: 私有内存分配（堆、栈）
4. **文件映射**: 以文件为后备存储的内存映射
5. **共享内存**: 进程间内存共享
6. **直接物理映射**: 设备内存映射

## 核心数据结构

### 1. 内存类型操作表 (mem_type_t)

**定义位置**: `servers/vm/memtype.h`

```c
typedef struct mem_type {
    const char *name;
    int (*ev_new)(struct vir_region *region);
    void (*ev_delete)(struct vir_region *region);
    int (*ev_reference)(struct phys_region *pr, struct phys_region *newpr);
    int (*ev_unreference)(struct phys_region *pr);
    int (*ev_pagefault)(struct vmproc *vmp, struct vir_region *region,
         struct phys_region *ph, int write, vfs_callback_t cb, void *state,
         int len, int *io);
    int (*ev_resize)(struct vmproc *vmp, struct vir_region *vr, vir_bytes len);
    void (*ev_split)(struct vmproc *vmp, struct vir_region *vr,
        struct vir_region *r1, struct vir_region *r2);
    int (*writable)(struct phys_region *pr);
    int (*ev_sanitycheck)(struct phys_region *pr, const char *file, int line);
    int (*ev_copy)(struct vir_region *vr, struct vir_region *newvr);
    int (*ev_lowshrink)(struct vir_region *vr, vir_bytes len);
    u32_t (*regionid)(struct vir_region *vr);
    int (*refcount)(struct vir_region *vr);
    int (*pt_flags)(struct vir_region *vr);
} mem_type_t;
```

**字段说明**:

| 字段 | 类型 | 说明 |
|------|------|------|
| `name` | `const char*` | 内存类型名称（用于调试） |
| `ev_new` | `int (*)(vir_region*)` | 创建区域事件 |
| `ev_delete` | `void (*)(vir_region*)` | 删除区域事件 |
| `ev_reference` | `int (*)(phys_region*, phys_region*)` | 引用物理区域事件 |
| `ev_unreference` | `int (*)(phys_region*)` | 取消引用物理区域事件 |
| `ev_pagefault` | `int (*)(...)` | 页错误处理事件 |
| `ev_resize` | `int (*)(vmproc*, vir_region*, vir_bytes)` | 调整区域大小事件 |
| `ev_split` | `void (*)(...)` | 分裂区域事件 |
| `writable` | `int (*)(phys_region*)` | 检查是否可写 |
| `ev_sanitycheck` | `int (*)(...)` | 健全性检查 |
| `ev_copy` | `int (*)(vir_region*, vir_region*)` | 复制区域事件 |
| `ev_lowshrink` | `int (*)(vir_region*, vir_bytes)` | 收缩区域事件 |
| `regionid` | `u32_t (*)(vir_region*)` | 获取区域 ID |
| `refcount` | `int (*)(vir_region*)` | 获取引用计数 |
| `pt_flags` | `int (*)(vir_region*)` | 获取页表标志 |

**设计原因**:
- **多态实现**: 使用函数指针实现内存类型多态
- **统一接口**: 所有内存类型使用相同的接口
- **扩展性**: 可以轻松添加新的内存类型

```
│                          mmap 流程                                           │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                             │

│                                                                             │

│                                    │                                        │
│                                    ↓                                        │
│   步骤 3: 创建区域                                                           │
│   ┌─────────────────────────────────────────────────────────────────────┐   │

│   │     if (!vr && (flags & MAP_FIXED))                                  │   │

│   │ }                                                                    │   │
│   │                                                                     │   │
│   │ if (!vr) {                                                           │   │
│   │     // 在 mmap 区域创建                                              │   │
│   │     vr = map_page_region(vmp, VM_MMAPBASE, VM_MMAPTOP, len,         │   │
│   │                          vrflags, mfflags, mt);                      │   │
│   │ }                                                                    │   │
│   └─────────────────────────────────────────────────────────────────────┘   │
│                                    │                                        │
│                                    ↓                                        │
│   步骤 4: 设置内存类型                                                       │
│   ┌─────────────────────────────────────────────────────────────────────┐   │
│   │ // 根据映射类型设置内存类型                                          │   │
│   │ if (flags & MAP_ANONYMOUS) {                                         │   │
│   │     // 匿名映射                                                      │   │
│   │     vr->def_memtype = &mem_type_anon;                                │   │
│   │ } else {                                                             │   │
│   │     // 文件映射                                                      │   │
│   │     vr->def_memtype = &mem_type_mappedfile;                          │   │
│   │     mappedfile_setfile(vmp, vr, fd, offset, dev, ino, ...);          │   │
│   │ }                                                                    │   │
│   └─────────────────────────────────────────────────────────────────────┘   │
│                                    │                                        │
│                                    ↓                                        │
│   返回区域起始地址                                                           │
│   return vr->vaddr;                                                         │
```

**关键代码**:

```c
static struct vir_region *mmap_region(struct vmproc *vmp, vir_bytes addr,
    u32_t vmm_flags, size_t len, u32_t vrflags,
    mem_type_t *mt, int execpriv)
{
    u32_t mfflags = 0;
    struct vir_region *vr = NULL;

    // 处理特殊标志
    if (vmm_flags & MAP_LOWER16M) vrflags |= VR_LOWER16MB;
    if (vmm_flags & MAP_LOWER1M)  vrflags |= VR_LOWER1MB;
    if (vmm_flags & MAP_ALIGNMENT_64KB) vrflags |= VR_PHYS64K;
    if (vmm_flags & MAP_PREALLOC) mfflags |= MF_PREALLOC;
    if (vmm_flags & MAP_UNINITIALIZED) {
        if (!execpriv) return NULL;
        vrflags |= VR_UNINITIALIZED;
    }

    // 验证长度
    if (len <= 0) {
        return NULL;
    }

    // 对齐到页边界
    if (len % VM_PAGE_SIZE)
        len += VM_PAGE_SIZE - (len % VM_PAGE_SIZE);

    // 处理 MAP_FIXED
    if (addr && (vmm_flags & MAP_FIXED)) {
        int r = map_unmap_range(vmp, addr, len);
        if (r != OK) {
            printf("mmap_region: map_unmap_range failed (%d)\n", r);
            return NULL;
        }
    }

    // 创建区域
    if (addr || (vmm_flags & MAP_FIXED)) {
        vr = map_page_region(vmp, addr, 0, len, vrflags, mfflags, mt);
        if (!vr && (vmm_flags & MAP_FIXED))
            return NULL;
    }

    if (!vr) {
        vr = map_page_region(vmp, VM_MMAPBASE, VM_MMAPTOP, len,
            vrflags, mfflags, mt);
    }

    return vr;
}
```

## 关键机制拆解

### 1. 内存类型多态

**原理**: 使用函数指针实现多态

## 边界条件与特殊分支

### 1. mmap 参数验证

**代码位置**: `mmap.c`

```c
if (len <= 0) {
    return NULL;
}

if (len % VM_PAGE_SIZE)
    len += VM_PAGE_SIZE - (len % VM_PAGE_SIZE);
```

**设计原因**: 长度必须为正且对齐到页边界。

### 2. MAP_FIXED 处理

**代码位置**: `mmap.c`

```c
if (addr && (flags & MAP_FIXED)) {
    int r = map_unmap_range(vmp, addr, len);
    if (r != OK) {
        printf("mmap_region: map_unmap_range failed (%d)\n", r);
        return NULL;
    }
}
```

**注意**: MAP_FIXED 要求在指定地址创建，需要先取消映射现有区域。

### 3. 文件映射缓存

**代码位置**: `mem_file.c`

```c
cp = find_cached_page_bydev(dev, offset, ...);

if (cp && (!cb || !(cp->flags & VMSF_ONCE))) {
    // 缓存命中
    pb_link(ph, cp->page, ph->offset, region);
    return OK;
}
```

**设计原因**: 使用缓存加速文件映射，减少 I/O。

### 4. 写时复制转换

**代码位置**: `mem_file.c`

```c
if (write) {
    cow_block(vmp, region, ph, clearend);

    ph->memtype = &mem_type_anon;
}
```

**设计原因**: 私有文件映射写操作后转换为匿名内存。

### 1. 类型系统改进

**当前问题**: 使用函数指针实现多态


```rust
trait MemoryType {

    fn handle_pagefault(
        &self,
        vmp: &mut VmProcess,
        region: &VirtualRegion,
        phys_region: &mut PhysRegion,

}

struct AnonymousMemory;
impl MemoryType for AnonymousMemory {
    fn name(&self) -> &'static str {
        "anonymous memory"
    }
    
    fn handle_pagefault(
        &self,
        vmp: &mut VmProcess,
        region: &VirtualRegion,
        phys_region: &mut PhysRegion,
        write: bool,
    ) -> Result<PageFaultResult, PageFaultError> {
        // 匿名内存页错误处理

            phys_region.phys = new_page;
            return Ok(PageFaultResult::Ok);

        if phys_region.refcount < 2 || !write {
            return Ok(PageFaultResult::Ok);
        }
        
        // 写时复制
        mem_cow(region, phys_region)?;
        Ok(PageFaultResult::Ok)
    }
    
    // ... 其他方法实现
}

struct FileMappedMemory {
    fd: FileDescriptor,
    offset: u64,
    dev: DeviceId,
    ino: InodeNumber,
}

impl MemoryType for FileMappedMemory {
    fn name(&self) -> &'static str {

        }
        
        Ok(PageFaultResult::Ok)
    }
    
    // ... 其他方法实现
}
```

### 4. 异步处理改进

```rust
use core::future::Future;

impl FileMappedMemory {
    async fn handle_pagefault_async(
        &self,
        vmp: &mut VmProcess,
        region: &VirtualRegion,
        phys_region: &mut PhysRegion,
        write: bool,

            if let Some(cached_page) = find_cached_page(self.dev, offset) {

            // 异步请求 VFS 读取
```

## 灾难预演

**场景 1: mmap 地址冲突**

如果 MAP_FIXED 指定的地址已被占用：
```c
if (addr && (flags & MAP_FIXED)) {
    r = map_unmap_range(vmp, addr, len);
    if (r != OK) {
        printf("mmap_region: map_unmap_range failed (%d)\n", r);
        return NULL;
    }
}
```
- 取消映射失败
- mmap 返回错误
- 进程需要处理错误

**预防**: 检查返回值，处理错误

**场景 2: 文件映射缓存失效**

如果缓存页被意外释放：
```c
if (cp) {
    pb_link(ph, cp->page, ph->offset, region);
    return OK;
}
```
- 缓存页被释放
- 访问无效内存
- 系统崩溃

**预防**: 使用引用计数保护缓存页

**场景 3: 共享内存源进程退出**

如果共享内存的源进程退出：
```c
if (vm_isokendpt(region->param.shared.ep, &srcproc) != OK) {
    printf("VM: shared memory with missing source process.\n");
    return EINVAL;
}
```
- 源进程已退出
- 共享内存失效
- 访问错误

**预防**: 检查源进程状态，处理错误

**场景 4: 直接物理映射地址错误**

如果直接物理映射的地址无效：
```c
phys_bytes phmem = region->param.phys + ph->offset;
ph->ph->phys = phmem;
```
- 物理地址无效
- 访问设备内存错误
- 设备故障

**预防**: 验证物理地址有效性
